cmake_minimum_required(VERSION 3.16)
project(pcl_sor LANGUAGES C CXX)
find_package(PCL REQUIRED COMPONENTS common io filters features)
add_executable(pcl_sor main.cpp)
add_executable(pcl_normals normals.cpp)
foreach(tool pcl_sor pcl_normals)
    target_compile_features(${tool} PRIVATE cxx_std_17)
    target_include_directories(${tool} PRIVATE ${PCL_INCLUDE_DIRS})
    target_link_directories(${tool} PRIVATE ${PCL_LIBRARY_DIRS})
    target_compile_options(${tool} PRIVATE ${PCL_DEFINITIONS})
    target_link_libraries(${tool} PRIVATE ${PCL_LIBRARIES})
endforeach()
