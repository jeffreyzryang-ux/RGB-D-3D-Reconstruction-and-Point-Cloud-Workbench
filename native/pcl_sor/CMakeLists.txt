cmake_minimum_required(VERSION 3.16)
project(pcl_sor LANGUAGES C CXX)
find_package(PCL REQUIRED COMPONENTS common io filters)
add_executable(pcl_sor main.cpp)
target_compile_features(pcl_sor PRIVATE cxx_std_17)
target_include_directories(pcl_sor PRIVATE ${PCL_INCLUDE_DIRS})
target_link_directories(pcl_sor PRIVATE ${PCL_LIBRARY_DIRS})
target_compile_options(pcl_sor PRIVATE ${PCL_DEFINITIONS})
target_link_libraries(pcl_sor PRIVATE ${PCL_LIBRARIES})
